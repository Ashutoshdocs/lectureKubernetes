apiVersion: v1
kind: Pod
metadata:
  name: multi-container-pod
spec:
  containers:
  - name: container1
    image: busybox
    command:
      - sh
      - -c
      - |
        while true
        do
          echo "Container1 Log: $(date)"
          sleep 5
        done

  - name: container2
    image: busybox
    command:
      - sh
      - -c
      - |
        while true
        do
          echo "Container2 Log: $(date)"
          sleep 3
        done
