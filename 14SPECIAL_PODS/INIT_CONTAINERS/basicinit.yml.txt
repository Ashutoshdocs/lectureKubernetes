apiVersion: v1
kind: Pod
metadata:
  name: init-demo
spec:

  volumes:
  - name: shared-data
    emptyDir: {}

  initContainers:
  - name: create-file
    image: busybox
    command:
    - sh
    - -c
    - |
      echo "Welcome from Init Container" > /data/welcome.txt

    volumeMounts:
    - name: shared-data
      mountPath: /data

  containers:
  - name: nginx
    image: nginx

    volumeMounts:
    - name: shared-data
      mountPath: /usr/share/nginx/html
